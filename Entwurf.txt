        // Initialize Cloud Storage (Firebase)
        async function initStorage() {
            try {
                const firebaseConfig = JSON.parse(typeof __firebase_config !== 'undefined' ? __firebase_config : '{}');
                appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
                const app = initializeApp(firebaseConfig);
                auth = getAuth(app);
                db = getFirestore(app);

                // RULE 3: Auth First
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }

                onAuthStateChanged(auth, (user) => {
                    if (user) {
                        currentUser = user;
                        setupListeners();
                        loadingOverlay.classList.add('opacity-0', 'pointer-events-none');
                    }
                });
            } catch (e) {
                console.error("Storage Error:", e);
                loadingOverlay.innerHTML = "<p class='text-red-500 font-bold'>Fehler beim Laden.</p>";
            }
        }

        // Listen for Database Changes
        function setupListeners() {
            if (!currentUser) return;

            // 1. Messages Listener (Private)
            const messagesRef = collection(db, 'artifacts', appId, 'users', currentUser.uid, 'messages');
            onSnapshot(messagesRef, (snapshot) => {
                const msgs = [];
                snapshot.forEach(d => msgs.push({ id: d.id, ...d.data() }));
                msgs.sort((a, b) => a.timestamp - b.timestamp);
                messagesState = msgs;
                if (msgs.length === 0) addInitialGreeting();
                else renderMessages();
            });

            // 2. Tickets Listener (Private)
            const ticketsRef = collection(db, 'artifacts', appId, 'users', currentUser.uid, 'tickets');
            onSnapshot(ticketsRef, (snapshot) => {
                const tks = [];
                snapshot.forEach(d => tks.push({ id: d.id, ...d.data() }));
                tks.sort((a, b) => b.timestamp - a.timestamp);
                ticketsState = tks;
                renderTickets();
            });

            // 3. Knowledge Base Listener (Public/App-wide)
            // Using public collection so the knowledge is shared/global for this app instance
            const knowledgeRef = collection(db, 'artifacts', appId, 'public', 'data', 'knowledge');
            onSnapshot(knowledgeRef, (snapshot) => {
                const kbase = [];
                snapshot.forEach(d => kbase.push({ id: d.id, ...d.data() }));
                kbase.sort((a, b) => b.timestamp - a.timestamp);
                knowledgeState = kbase;
                renderKnowledge();
            });
        }

        // UI Renderers
        async function addInitialGreeting() {

            await addDoc(collection(db, 'artifacts', appId, 'users', currentUser.uid, 'messages'), {
                text: text, sender: 'bot', isMarkdown: true, timestamp: Date.now()
            });
        }

        function renderMessages() {
            chatContainer.innerHTML = `<div class="flex justify-center mb-4"><span class="text-xs text-gray-400 bg-gray-200 px-3 py-1 rounded-full">Gespeicherter Verlauf</span></div>`;

                    return;
                }
                if(msg.type === 'learned_ui') {
                    chatContainer.insertAdjacentHTML('beforeend', generateLearnedUIHTML(msg.problem));
                    return;
                }

                const time = new Date(msg.timestamp).toLocaleTimeString('de-DE', { hour: '2-digit', minute: '2-digit' });
                let messageContent = msg.isMarkdown ? marked.parse(msg.text) : msg.text.replace(/\n/g, '<br>');
                const isUser = msg.sender === 'user';
                
                // Color formatting based on context (if bot replied in teacher mode, make it slightly violet)
                const botBg = msg.isTeacherContext ? 'bg-violet-50 border border-violet-200 text-violet-900' : 'bg-white border border-gray-200 text-gray-700';

        // Gemini API Builder
        async function fetchGeminiResponse() {
            // 1. Compile the Knowledge Base into the prompt
            let knowledgeString = "";
            if (knowledgeState.length > 0) {
                knowledgeString = "\n\nDEINE WISSENSDATENBANK (WICHTIG: Nutze diese Lösungen zwingend, wenn das Problem dazu passt!):\n";
                knowledgeState.forEach(k => {
                    knowledgeString += `- WENN PROBLEM: "${k.problem}" DANN LÖSUNG: "${k.solution}"\n`;
                });
            } else {
                knowledgeString = "\n\nDeine Wissensdatenbank ist aktuell leer.";
            }

            // 2. Adjust prompt based on role
            let roleInstructions = "";
            if (currentRole === 'teacher') {
                roleInstructions = `
Du sprichst aktuell mit einem LEHRER / ADMINISTRATOR. 
Er darf dir neues Wissen beibringen. 
WENN er dir eine Anweisung gibt wie "Merke dir...", "Lerne Folgendes..." oder dir eine neue Lösung für ein Problem nennt:
1. Bedanke dich für das neue Wissen.
2. ZWINGEND: Am GANZ ENNEN deiner Antwort musst du exakt diesen Tag anfügen: [LEARNED: Kurze Problembeschreibung ||| Detaillierte Lösung]
Beispiel-Ausgabe am Ende: [LEARNED: Smartboard flackert ||| HDMI-Kabel an Port 2 umstecken]`;
            } else {
                roleInstructions = `
Du sprichst aktuell mit einem SCHÜLER. Er darf KEIN neues Wissen anlegen. 
Wenn du keine Lösung in der Wissensdatenbank findest, biete an ein Ticket zu erstellen.
WENN ein Ticket erstellt werden soll, schreibe ZWINGEND am Ende deiner Antwort: [TICKET_CREATED]`;
            }

            const basePrompt = `Du bist ein freundlicher IT-Support-Chatbot für eine Schule. Halte deine Antworten kurz und übersichtlich.`;
            const finalPrompt = basePrompt + knowledgeString + "\n\n" + roleInstructions;

            // 3. Build Chat History
            const contents = messagesState
                .filter(m => m.type !== 'ticket_ui' && m.type !== 'learned_ui')
                .slice(-10) // Only send last 10 messages to save tokens
                .map(msg => ({
                    role: msg.sender === 'user' ? 'user' : 'model',
                    parts: [{ text: msg.text }]
                }));

            const payload = {
                contents: contents,
                systemInstruction: { parts: [{ text: finalPrompt }] }
            };

            try {
                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                if (!response.ok) throw new Error('API Request failed');
                const data = await response.json();
                return data.candidates[0].content.parts[0].text;
            } catch (error) {
                console.error("Gemini Error:", error);
                return "Entschuldigung, meine Serververbindung klemmt gerade.";
            }
        }

        // Handle Send
        async function handleSend() {
            const text = inputField.value.trim();
            if (!text || isWaitingForResponse || !currentUser) return;

            // UI Reset
            inputField.value = '';
            inputField.style.height = '40px';
            
            // Save User Message
            await addDoc(collection(db, 'artifacts', appId, 'users', currentUser.uid, 'messages'), {
                text: text, sender: 'user', isMarkdown: false, timestamp: Date.now()
            });

            // Loading state
            isWaitingForResponse = true;
            sendBtn.disabled = true;
            typingContainer.classList.remove('hidden');
            chatContainer.scrollTop = chatContainer.scrollHeight;

            // Fetch AI
            let aiResponse = await fetchGeminiResponse();

            typingContainer.classList.add('hidden');
            isWaitingForResponse = false;
            sendBtn.disabled = false;
            
            // --- Parse Actions from AI ---
            let createTicket = false;
            let learnedKnowledge = null;

            if (aiResponse.includes('[TICKET_CREATED]')) {
                createTicket = true;
                aiResponse = aiResponse.replace('[TICKET_CREATED]', '').trim();
            }

            // Regex to find [LEARNED: problem ||| solution]
            const learnRegex = /\[LEARNED:\s*(.*?)\s*\|\|\|\s*(.*?)\]/s;
            const learnMatch = aiResponse.match(learnRegex);
            
            if (learnMatch && currentRole === 'teacher') {
                learnedKnowledge = {
                    problem: learnMatch[1].trim(),
                    solution: learnMatch[2].trim()
                };
                aiResponse = aiResponse.replace(learnRegex, '').trim(); // Remove the tag from output
            }

            // Save Bot Message
            await addDoc(collection(db, 'artifacts', appId, 'users', currentUser.uid, 'messages'), {
                text: aiResponse, 
                sender: 'bot', 
                isMarkdown: true, 
                timestamp: Date.now(),
                isTeacherContext: currentRole === 'teacher' // Style differently if teacher
            });

            // Handle Ticket Action
            if (createTicket) {
                const newTicketId = `#TK-${Math.floor(1000 + Math.random() * 9000)}`;
                await addDoc(collection(db, 'artifacts', appId, 'users', currentUser.uid, 'tickets'), {
                    ticketId: newTicketId, title: "Chat-Eskalation", timestamp: Date.now()
                });
                await addDoc(collection(db, 'artifacts', appId, 'users', currentUser.uid, 'messages'), {
                    type: 'ticket_ui', ticketId: newTicketId, timestamp: Date.now() + 1
                });
            }

            // Handle Learn Action
            if (learnedKnowledge) {
                // Save to public knowledge base
                await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'knowledge'), {
                    problem: learnedKnowledge.problem,
                    solution: learnedKnowledge.solution,
                    timestamp: Date.now()
                });
                // Add visual confirmation in chat
                await addDoc(collection(db, 'artifacts', appId, 'users', currentUser.uid, 'messages'), {
                    type: 'learned_ui', problem: learnedKnowledge.problem, timestamp: Date.now() + 2
                });
            }

            inputField.focus();
        }

        // Clear Chat History
        clearChatBtn.addEventListener('click', async () => {
            if(!currentUser) return;
            for (const msg of messagesState) {
                await deleteDoc(doc(db, 'artifacts', appId, 'users', currentUser.uid, 'messages', msg.id));
            }
        });

        // Event Listeners
        sendBtn.addEventListener('click', handleSend);
        inputField.addEventListener('keypress', function(e) {
            if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); handleSend(); }
        });
        inputField.addEventListener('input', function() {
            this.style.height = '40px'; this.style.height = (this.scrollHeight) + 'px';
        });

        initStorage();
    </script>
